\begin{questaoManually}{Exercício 1.1}

\noindent\textbf{Função de entrada:}
\[
  z = \mbf{w}^\top \mbf{x} + b = \sum_{i=1}^{d} w_i\, x_i + b
\]

\medskip

\noindent\textbf{Função de ativação degrau unitário:}
\[
  \phi(z) =
  \begin{cases}
    +1, & \text{se } z \geq 0, \\
    -1, & \text{se } z < 0.
  \end{cases}
\]

\medskip

\noindent\textbf{Regra de decisão final:}
\[
  \hat{y} = f(\mbf{x}) = \phi\!\left(\mbf{w}^\top \mbf{x} + b\right) =
  \begin{cases}
    +1, & \text{se } \mbf{w}^\top \mbf{x} + b \geq 0, \\
    -1, & \text{se } \mbf{w}^\top \mbf{x} + b < 0.
  \end{cases}
\]

\end{questaoManually}
\begin{questaoManually}{Exercício 1.4}

Sejam $\mbf{w} = [0.4,\,-0.6]^\top$ e $b = 0.1$.

\medskip

\noindent\textbf{Instância 1:} $\mbf{x}^{(1)} = [2.0,\,1.0]^\top$

\[
  z^{(1)} = \mbf{w}^\top \mbf{x}^{(1)} + b = 0.4 \cdot 2.0 + (-0.6) \cdot 1.0 + 0.1 = 0.8 - 0.6 + 0.1 = 0.3
\]

\[
  \hat{y}^{(1)} = \phi(0.3) = +1 \quad (\text{pois } z \geq 0)
\]

\medskip

\noindent\textbf{Instância 2:} $\mbf{x}^{(2)} = [1.0,\,2.0]^\top$

\[
  z^{(2)} = \mbf{w}^\top \mbf{x}^{(2)} + b = 0.4 \cdot 1.0 + (-0.6) \cdot 2.0 + 0.1 = 0.4 - 1.2 + 0.1 = -0.7
\]

\[
  \hat{y}^{(2)} = \phi(-0.7) = -1 \quad (\text{pois } z < 0)
\]

\end{questaoManually}
