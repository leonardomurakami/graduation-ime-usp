\begin{exercicio}
  \begin{enunciado}
    Encontre o valor esperado e a variância, utilizando os atalhos criados.
  \end{enunciado}

  Usando os nossos novos atalhos matemáticos:
  \begin{equation*}
    \E[X] = \sum_{i=1}^{n} x_i p_i \quad \text{e} \quad \Var(X) = \E[X^2] - (\E[X])^2
  \end{equation*}
  
  Também podemos usar $\argmax_{x} f(x)$ facilmente.
\end{exercicio}

\begin{exercicio}[(Busca A*)]
  Exemplo de como o ambiente de algoritmo funciona agora:
  
  \begin{algorithm}[H]
    \caption{Busca A* Simplificada}
    \Fn{AStar($inicio, objetivo$)}{
      $aberta \gets \{inicio\}$\;
      $fechada \gets \emptyset$\;
      \Enqto{$aberta \neq \emptyset$}{
        $atual \gets$ nó em $aberta$ com menor $f(n)$\;
        \Se{$atual == objetivo$}{
          \Retorna caminho\;
        }
        Mova $atual$ para $fechada$\;
        Expanda vizinhos de $atual$\;
      }
      \Retorna \False\;
    }
  \end{algorithm}
\end{exercicio}
